``` python
from pathlib import Path
import re
import sys

# ============================================================
# 三层双语字幕 v3.2
#
# 核心：
# 1. SRT 第一行英文、第二行中文
# 2. 英文自动拆成上下两部分
# 3. 中文在上，英文两层在下
# 4. 英文第一部分较淡，第二部分较亮
# 5. 真正计算文字宽度，使用 ASS 绘制独立背景矩形
# 6. BOX_PADDING_X / BOX_MIN_WIDTH 真正以像素控制
#
# 注意：
# 本版本不再使用 \h 空格“撑”字幕框。
# 字幕文字和背景框分开绘制，因此宽度控制是真实的。
# ============================================================

VIDEO_WIDTH = 1920
VIDEO_HEIGHT = 1080

# ---------- 字体 ----------
ZH_FONT = "Microsoft YaHei"
EN_FONT = "Arial"

ZH_SIZE = 32
EN_SIZE = 46

# ---------- 颜色 ----------
ZH_COLOR = "#B8B8B8"
EN1_COLOR = "#B8B8B8"
EN2_COLOR = "#FFFFFF"

# ---------- 字幕框 ----------
BOX_COLOR = "#111111"
BOX_ALPHA = 35
OUTLINE_COLOR = "#000000"

# 真正的左右像素留白
BOX_PADDING_X = 30

# 字幕框最小宽度
BOX_MIN_WIDTH = 100

# 字幕框高度
BOX_HEIGHT_ZH = 54
BOX_HEIGHT_EN = 64

# 字幕框圆角半径（像素）
# 0 = 直角；20~30 = 轻微圆角；40~50 = 明显圆角
BOX_RADIUS = 20

# ---------- 三层位置 ----------
ZH_Y = 50
EN1_Y = 975
EN2_Y = 1025

# ---------- 字体宽度估算 ----------
# Arial / Microsoft YaHei 的实际渲染宽度会因字符不同而变化。
# 这里使用经验系数作为 ASS 位置框的近似计算。
ZH_CHAR_WIDTH = 0.95
EN_CHAR_WIDTH = 0.56

MARGIN_L = 100
MARGIN_R = 100


def ass_color(hex_color, transparency=0):
    c = hex_color.lstrip("#")
    r, g, b = c[0:2], c[2:4], c[4:6]
    alpha = round(max(0, min(100, transparency)) * 255 / 100)
    return f"&H{alpha:02X}{b}{g}{r}".upper()


def ass_time(t):
    t = t.replace(",", ".")
    h, m, rest = t.split(":")
    s, ms = rest.split(".")

    total_cs = round(int(ms) / 10)
    total_seconds = int(h) * 3600 + int(m) * 60 + int(s)

    if total_cs >= 100:
        total_seconds += total_cs // 100
        total_cs %= 100

    hours = total_seconds // 3600
    minutes = (total_seconds % 3600) // 60
    seconds = total_seconds % 60

    return f"{hours}:{minutes:02d}:{seconds:02d}.{total_cs:02d}"


def clean_text(text):
    text = re.sub(r"<[^>]+>", "", text)
    text = text.replace("\u200b", "")
    return " ".join(text.split())


def split_english(text):
    text = clean_text(text)

    if not text:
        return "", ""

    sentence_parts = re.split(r"(?<=[.!?])\s+", text)

    if len(sentence_parts) >= 2:
        best = None
        best_diff = None

        for i in range(1, len(sentence_parts)):
            a = " ".join(sentence_parts[:i]).strip()
            b = " ".join(sentence_parts[i:]).strip()

            if not a or not b:
                continue

            diff = abs(len(a) - len(b))

            if best_diff is None or diff < best_diff:
                best = (a, b)
                best_diff = diff

        if best:
            return best

    words = text.split()

    if len(words) <= 1:
        return text, ""

    total = len(text)
    target = total / 2

    current = 0
    split_at = 1
    best_distance = float("inf")

    for i in range(1, len(words)):
        current += len(words[i - 1]) + 1
        distance = abs(current - target)

        if distance < best_distance:
            best_distance = distance
            split_at = i

    return (
        " ".join(words[:split_at]).strip(),
        " ".join(words[split_at:]).strip()
    )


def parse_srt(text):
    result = []

    blocks = re.split(r"\r?\n\s*\r?\n", text.strip())

    for block in blocks:
        raw_lines = [
            x.strip("\ufeff").strip()
            for x in block.splitlines()
            if x.strip()
        ]

        if len(raw_lines) < 3:
            continue

        match = None
        time_index = None

        for i, raw_line in enumerate(raw_lines):
            m = re.match(
                r"(\d\d:\d\d:\d\d[,.]\d{3})\s*-->\s*"
                r"(\d\d:\d\d:\d\d[,.]\d{3})",
                raw_line
            )
            if m:
                match = m
                time_index = i
                break

        if not match:
            continue

        subtitle_lines = [
            clean_text(line)
            for line in raw_lines[time_index + 1:]
        ]

        if len(subtitle_lines) < 2:
            continue

        english = subtitle_lines[0]
        chinese = subtitle_lines[1]

        en1, en2 = split_english(english)

        result.append({
            "start": ass_time(match.group(1)),
            "end": ass_time(match.group(2)),
            "zh": chinese,
            "en1": en1,
            "en2": en2,
        })

    return result


def esc(text):
    return (
        text.replace("\\", r"\\")
            .replace("{", r"\{")
            .replace("}", r"\}")
    )


def estimate_text_width(text, font_size, char_width):
    """
    估算字幕文字宽度。

    对英文按字符分别估算：
    - 空格更窄
    - 大写略宽
    - 标点较窄

    对中文按字符估算。
    """
    width = 0.0

    for ch in text:
        if ch == " ":
            width += font_size * 0.28
        elif ord(ch) >= 0x2E80:
            width += font_size * char_width
        elif ch in ".,!?;:'\"-()[]":
            width += font_size * 0.30
        elif ch.isupper():
            width += font_size * char_width * 1.08
        else:
            width += font_size * char_width

    return max(1, round(width))


def box_width(text, font_size, char_width):
    text_width = estimate_text_width(
        text,
        font_size,
        char_width
    )

    return max(
        BOX_MIN_WIDTH,
        text_width + BOX_PADDING_X * 2
    )


def dialogue_text(
    layer,
    start,
    end,
    style,
    text,
    x,
    y,
):
    if not text:
        return None

    return (
        f"Dialogue: {layer},{start},{end},{style},"
        f",0,0,0,,{{\\pos({x},{y})}}{esc(text)}"
    )


def dialogue_box(
    layer,
    start,
    end,
    style,
    width,
    height,
    x,
    y,
):
    """
    使用 ASS Drawing Mode 绘制圆角矩形背景。

    ASS 的 b 命令用于三次 Bézier 曲线。
    这里用 0.5522848 的控制点近似四个 90° 圆弧。
    """

    width = max(1, round(width))
    height = max(1, round(height))

    radius = max(0, min(
        round(BOX_RADIUS),
        width // 2,
        height // 2
    ))

    left = round(x - width / 2)
    top = round(y - height / 2)
    right = round(x + width / 2)
    bottom = round(y + height / 2)

    if radius == 0:
        drawing = (
            f"{{\\p1}}"
            f"m {left} {top} "
            f"l {right} {top} "
            f"l {right} {bottom} "
            f"l {left} {bottom} "
            f"l {left} {top}"
            f"{{\\p0}}"
        )
    else:
        # 三次 Bézier 圆角控制点系数
        k = 0.5522848
        c = round(radius * k)

        drawing = (
            f"{{\\p1}}"
            # 从顶部左侧圆角结束点开始
            f"m {left + radius} {top} "
            # 顶边
            f"l {right - radius} {top} "
            # 右上圆角
            f"b {right - radius + c} {top} "
            f"{right} {top + radius - c} "
            f"{right} {top + radius} "
            # 右边
            f"l {right} {bottom - radius} "
            # 右下圆角
            f"b {right} {bottom - radius + c} "
            f"{right - radius + c} {bottom} "
            f"{right - radius} {bottom} "
            # 底边
            f"l {left + radius} {bottom} "
            # 左下圆角
            f"b {left + radius - c} {bottom} "
            f"{left} {bottom - radius + c} "
            f"{left} {bottom - radius} "
            # 左边
            f"l {left} {top + radius} "
            # 左上圆角
            f"b {left} {top + radius - c} "
            f"{left + radius - c} {top} "
            f"{left + radius} {top}"
            f"{{\\p0}}"
        )

    return (
        f"Dialogue: {layer},{start},{end},{style},"
        f",0,0,0,,{drawing}"
    )


def make_ass(items, output):
    zh_color = ass_color(ZH_COLOR)
    en1_color = ass_color(EN1_COLOR)
    en2_color = ass_color(EN2_COLOR)
    outline = ass_color(OUTLINE_COLOR)
    box = ass_color(BOX_COLOR, BOX_ALPHA)

    # 背景矩形专用样式。
    # Alignment=7 只是为了让绘图层保持稳定，实际坐标由 m/l 控制。
    header = f"""[Script Info]
ScriptType: v4.00+
PlayResX: {VIDEO_WIDTH}
PlayResY: {VIDEO_HEIGHT}
WrapStyle: 2
ScaledBorderAndShadow: yes

[V4+ Styles]
Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding
Style: Chinese,{ZH_FONT},{ZH_SIZE},{zh_color},{zh_color},{outline},{box},-1,0,0,0,100,100,0,0,1,1,0,5,{MARGIN_L},{MARGIN_R},0,1
Style: English1,{EN_FONT},{EN_SIZE},{en1_color},{en1_color},{outline},{box},-1,0,0,0,100,100,0,0,1,1,0,5,{MARGIN_L},{MARGIN_R},0,1
Style: English2,{EN_FONT},{EN_SIZE},{en2_color},{en2_color},{outline},{box},-1,0,0,0,100,100,0,0,1,1,0,5,{MARGIN_L},{MARGIN_R},0,1
Style: BoxChinese,{ZH_FONT},{ZH_SIZE},{box},{box},{box},{box},0,0,0,0,100,100,0,0,1,0,0,7,0,0,0,1
Style: BoxEnglish1,{EN_FONT},{EN_SIZE},{box},{box},{box},{box},0,0,0,0,100,100,0,0,1,0,0,7,0,0,0,1
Style: BoxEnglish2,{EN_FONT},{EN_SIZE},{box},{box},{box},{box},0,0,0,0,100,100,0,0,1,0,0,7,0,0,0,1

[Events]
Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text
"""

    lines = [header]

    for item in items:
        start = item["start"]
        end = item["end"]

        # -------------------------
        # 中文
        # -------------------------
        zh = item["zh"]

        if zh:
            width = box_width(
                zh,
                ZH_SIZE,
                ZH_CHAR_WIDTH
            )

            lines.append(
                dialogue_box(
                    0,
                    start,
                    end,
                    "BoxChinese",
                    width,
                    BOX_HEIGHT_ZH,
                    VIDEO_WIDTH // 2,
                    ZH_Y,
                )
            )

            lines.append(
                dialogue_text(
                    1,
                    start,
                    end,
                    "Chinese",
                    zh,
                    VIDEO_WIDTH // 2,
                    ZH_Y,
                )
            )

        # -------------------------
        # 英文第一部分
        # -------------------------
        en1 = item["en1"]

        if en1:
            width = box_width(
                en1,
                EN_SIZE,
                EN_CHAR_WIDTH
            )

            lines.append(
                dialogue_box(
                    0,
                    start,
                    end,
                    "BoxEnglish1",
                    width,
                    BOX_HEIGHT_EN,
                    VIDEO_WIDTH // 2,
                    EN1_Y,
                )
            )

            lines.append(
                dialogue_text(
                    1,
                    start,
                    end,
                    "English1",
                    en1,
                    VIDEO_WIDTH // 2,
                    EN1_Y,
                )
            )

        # -------------------------
        # 英文第二部分
        # -------------------------
        en2 = item["en2"]

        if en2:
            width = box_width(
                en2,
                EN_SIZE,
                EN_CHAR_WIDTH
            )

            lines.append(
                dialogue_box(
                    0,
                    start,
                    end,
                    "BoxEnglish2",
                    width,
                    BOX_HEIGHT_EN,
                    VIDEO_WIDTH // 2,
                    EN2_Y,
                )
            )

            lines.append(
                dialogue_text(
                    1,
                    start,
                    end,
                    "English2",
                    en2,
                    VIDEO_WIDTH // 2,
                    EN2_Y,
                )
            )

    output.write_text(
        "\n".join(lines),
        encoding="utf-8-sig"
    )


def main():
    if len(sys.argv) < 2:
        print(
            "用法：python make_ass_final_v3_2.py "
            "input.srt [output.ass]"
        )
        raise SystemExit(1)

    input_path = Path(sys.argv[1])

    if not input_path.exists():
        print(f"找不到文件：{input_path}")
        raise SystemExit(1)

    output_path = (
        Path(sys.argv[2])
        if len(sys.argv) >= 3
        else input_path.with_name(
            input_path.stem + "_final.ass"
        )
    )

    raw = input_path.read_bytes()

    for encoding in ("utf-8-sig", "utf-8", "gb18030"):
        try:
            text = raw.decode(encoding)
            break
        except UnicodeDecodeError:
            continue
    else:
        print("无法识别 SRT 编码。")
        raise SystemExit(1)

    items = parse_srt(text)

    if not items:
        print("没有解析到有效字幕。")
        raise SystemExit(1)

    make_ass(items, output_path)

    print()
    print("========================================")
    print("三层字幕 v3.2 生成完成")
    print("========================================")
    print(f"输入：{input_path}")
    print(f"输出：{output_path}")
    print(f"字幕块：{len(items)}")
    print()
    print("字幕框参数：")
    print(f"  左右留白：{BOX_PADDING_X}px")
    print(f"  最小宽度：{BOX_MIN_WIDTH}px")
    print(f"  圆角半径：{BOX_RADIUS}px")
    print()
    print("处理逻辑：")
    print("  原 SRT：英文 + 中文")
    print("  ↓")
    print("  英文自动拆成上下两部分")
    print("  ↓")
    print("  中文最上")
    print("  英文第一部分：较淡")
    print("  英文第二部分：较亮")
    print("  ↓")
    print("  单独计算文字宽度")
    print("  ↓")
    print("  独立绘制真实字幕背景框")
    print("========================================")


if __name__ == "__main__":
    main()
```

``` python
from pathlib import Path
import re
import sys

VIDEO_WIDTH, VIDEO_HEIGHT = 1920, 1080
ZH_FONT, EN_FONT = "Microsoft YaHei", "Arial"
ZH_SIZE, EN_SIZE = 34, 42

def ass_time(value):
    h, m, rest = value.replace(",", ".").split(":")
    s, ms = rest.split(".")
    cs = round(int(ms) / 10)
    if cs >= 100:
        cs, s = 0, str(int(s) + 1)
    return f"{int(h)}:{int(m):02d}:{int(s):02d}.{cs:02d}"

def clean_text(s):
    return re.sub(r"\s+", " ", re.sub(r"<[^>]+>", "", s)).strip()

def parse_srt(text):
    result = []
    for block in re.split(r"\r?\n\s*\r?\n", text.strip()):
        lines = [x.strip("\ufeff").strip() for x in block.splitlines() if x.strip()]
        if len(lines) < 3:
            continue
        match = None
        ti = None
        for i, line in enumerate(lines[:3]):
            m = re.match(r"(\d{2}:\d{2}:\d{2}[,.]\d{3})\s*-->\s*(\d{2}:\d{2}:\d{2}[,.]\d{3})", line)
            if m:
                match, ti = m, i
                break
        if not match:
            continue
        sub = [clean_text(x) for x in lines[ti + 1:] if clean_text(x)]
        if not sub:
            continue
        result.append({
            "start": ass_time(match.group(1)),
            "end": ass_time(match.group(2)),
            "english": sub[0],
            "chinese": sub[1] if len(sub) > 1 else ""
        })
    return result

def split_chinese(text):
    text = text.strip()
    if not text:
        return "", ""
    if "\n" in text:
        p = [x.strip() for x in text.splitlines() if x.strip()][:2]
        return p[0], p[1] if len(p) > 1 else ""
    if len(text) <= 16:
        return text, ""
    punct = "，。！？；：、,.!?;:"
    target = len(text) / 2
    candidates = [(abs(i + 1 - target), i + 1) for i, c in enumerate(text)
                  if c in punct and 0 < i < len(text) - 1]
    pos = min(candidates)[1] if candidates else len(text) // 2
    return text[:pos].strip(), text[pos:].strip()

def split_english(text):
    text = re.sub(r"\s+", " ", text.strip())
    if not text:
        return "", "", ""
    if "\n" in text:
        p = [x.strip() for x in text.splitlines() if x.strip()][:3]
        return tuple(p + [""] * (3 - len(p)))
    words = text.split()
    if len(text) <= 42:
        return text, "", ""
    lines, cur = [], ""
    for word in words:
        candidate = word if not cur else cur + " " + word
        if len(candidate) <= 42:
            cur = candidate
        else:
            if cur:
                lines.append(cur)
            cur = word
    if cur:
        lines.append(cur)
    if len(lines) > 3:
        lines = lines[:2] + [" ".join(lines[2:])]
    lines += [""] * (3 - len(lines))
    return lines[:3]

def esc(s):
    return s.replace("\\", r"\\").replace("{", r"\{").replace("}", r"\}")

def write_ass(path, style, font, size, rows):
    # 关键：没有 \\pos()、\\move()、固定 X/Y。
    # 只有普通 ASS Style，方便字幕软件自行定位。
    header = f"""[Script Info]
ScriptType: v4.00+
PlayResX: {VIDEO_WIDTH}
PlayResY: {VIDEO_HEIGHT}
WrapStyle: 2
ScaledBorderAndShadow: yes

[V4+ Styles]
Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding
Style: {style},{font},{size},&H00FFFFFF,&H00FFFFFF,&H00000000,&H00000000,-1,0,0,0,100,100,0,0,1,2,0,2,40,40,20,1

[Events]
Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text
"""
    events = [
        f"Dialogue: 0,{start},{end},{style},,,0,0,0,,{esc(text)}"
        for start, end, text in rows if text
    ]
    path.write_text(header + "\n".join(events) + "\n", encoding="utf-8-sig")

def main():
    if len(sys.argv) < 2:
        print("用法：python make_ass_5track_free.py input.srt")
        raise SystemExit(1)
    inp = Path(sys.argv[1])
    raw = inp.read_bytes()
    for enc in ("utf-8-sig", "utf-8", "gb18030"):
        try:
            text = raw.decode(enc)
            break
        except UnicodeDecodeError:
            pass
    else:
        raise RuntimeError("无法识别 SRT 编码")

    items = parse_srt(text)
    if not items:
        raise RuntimeError("没有解析到有效 SRT 字幕")

    outdir = inp.parent / f"{inp.stem}_5track_free"
    outdir.mkdir(exist_ok=True)

    tracks = {k: [] for k in ("Cn1.ass","Cn2.ass","En1.ass","En2.ass","En3.ass")}
    for x in items:
        zh1, zh2 = split_chinese(x["chinese"])
        en1, en2, en3 = split_english(x["english"])
        tracks["Cn1.ass"].append((x["start"], x["end"], zh1))
        tracks["Cn2.ass"].append((x["start"], x["end"], zh2))
        tracks["En1.ass"].append((x["start"], x["end"], en1))
        tracks["En2.ass"].append((x["start"], x["end"], en2))
        tracks["En3.ass"].append((x["start"], x["end"], en3))

    cfg = {
        "Cn1.ass": ("Cn1", ZH_FONT, ZH_SIZE),
        "Cn2.ass": ("Cn2", ZH_FONT, ZH_SIZE),
        "En1.ass": ("En1", EN_FONT, EN_SIZE),
        "En2.ass": ("En2", EN_FONT, EN_SIZE),
        "En3.ass": ("En3", EN_FONT, EN_SIZE),
    }
    for name, rows in tracks.items():
        write_ass(outdir / name, *cfg[name], rows)

    print("完成")
    print(f"输入：{inp}")
    print(f"字幕块：{len(items)}")
    print(f"输出目录：{outdir}")
    print("已生成 Cn1.ass / Cn2.ass / En1.ass / En2.ass / En3.ass")
    print("本版没有写入 POS/MOVE 固定坐标。")

if __name__ == "__main__":
    main()
```

``` python
from pathlib import Path
import re
import sys

# ============================================================
# 五轨 SRT 拆分版
#
# Python 只负责：
#   1. 读取原 SRT
#   2. 保留原时间轴
#   3. 中文拆成 Cn1 / Cn2
#   4. 英文拆成 En1 / En2 / En3
#
# SRT 本身不写位置、字体、坐标、字幕框等信息。
# 最终位置完全交给字幕软件。
# ============================================================


def clean_text(text):
    # 删除 HTML 标签，保留普通字幕文字
    text = re.sub(r"<[^>]+>", "", text)
    return text.strip()


def parse_srt(text):
    result = []

    blocks = re.split(r"\r?\n\s*\r?\n", text.strip())

    for block in blocks:
        lines = [
            x.strip("\ufeff").strip()
            for x in block.splitlines()
            if x.strip()
        ]

        if len(lines) < 3:
            continue

        # 找时间轴，不强制要求它一定在第二行
        time_index = None
        start = None
        end = None

        for i, line in enumerate(lines[:3]):
            m = re.match(
                r"(\d{2}:\d{2}:\d{2}[,.]\d{3})\s*-->\s*"
                r"(\d{2}:\d{2}:\d{2}[,.]\d{3})",
                line
            )

            if m:
                time_index = i
                start = m.group(1).replace(".", ",")
                end = m.group(2).replace(".", ",")
                break

        if time_index is None:
            continue

        subtitle_lines = [
            clean_text(x)
            for x in lines[time_index + 1:]
            if clean_text(x)
        ]

        if not subtitle_lines:
            continue

        # 当前双语 SRT：
        #
        # English
        # 中文
        #
        # 因此：
        # subtitle_lines[0] = English
        # subtitle_lines[1] = Chinese

        english = subtitle_lines[0]
        chinese = subtitle_lines[1] if len(subtitle_lines) > 1 else ""

        result.append({
            "start": start,
            "end": end,
            "english": english,
            "chinese": chinese,
        })

    return result


def split_chinese(text):
    """
    中文最多两行。
    尽量在标点处断开。
    """

    text = text.strip()

    if not text:
        return "", ""

    if "\n" in text:
        parts = [x.strip() for x in text.splitlines() if x.strip()]
        return (
            parts[0],
            parts[1] if len(parts) > 1 else ""
        )

    # 短句不拆
    if len(text) <= 16:
        return text, ""

    punctuation = "，。！？；：、,.!?;:"
    target = len(text) / 2

    candidates = []

    for i, char in enumerate(text):
        if char in punctuation and 0 < i < len(text) - 1:
            candidates.append(
                (abs((i + 1) - target), i + 1)
            )

    if candidates:
        position = min(candidates)[1]
    else:
        position = len(text) // 2

    return (
        text[:position].strip(),
        text[position:].strip()
    )


def split_english(text):
    """
    英文最多三行。
    按单词拆分，不在单词中间切断。
    """

    text = re.sub(r"\s+", " ", text.strip())

    if not text:
        return "", "", ""

    # 原本已有换行
    if "\n" in text:
        parts = [
            x.strip()
            for x in text.splitlines()
            if x.strip()
        ]

        parts = parts[:3]

        return (
            parts[0] if len(parts) > 0 else "",
            parts[1] if len(parts) > 1 else "",
            parts[2] if len(parts) > 2 else "",
        )

    # 比较短，不拆
    if len(text) <= 42:
        return text, "", ""

    words = text.split()

    lines = []
    current = ""

    for word in words:

        candidate = (
            word
            if not current
            else current + " " + word
        )

        if len(candidate) <= 42:
            current = candidate

        else:
            if current:
                lines.append(current)

            current = word

    if current:
        lines.append(current)

    # 最多三行
    if len(lines) > 3:
        lines = [
            lines[0],
            lines[1],
            " ".join(lines[2:])
        ]

    while len(lines) < 3:
        lines.append("")

    return lines[0], lines[1], lines[2]


def write_srt(path, rows):
    """
    写普通 SRT。
    没有任何 ASS 样式或位置数据。
    """

    output = []

    number = 1

    for start, end, text in rows:

        if not text:
            continue

        output.append(str(number))
        output.append(
            f"{start} --> {end}"
        )
        output.append(text)
        output.append("")

        number += 1

    path.write_text(
        "\n".join(output),
        encoding="utf-8-sig"
    )


def main():

    if len(sys.argv) < 2:
        print(
            "用法：python split_srt_5track.py input.srt"
        )
        raise SystemExit(1)

    input_path = Path(sys.argv[1])

    if not input_path.exists():
        print(f"找不到文件：{input_path}")
        raise SystemExit(1)

    raw = input_path.read_bytes()

    # 兼容常见中文 SRT 编码
    for encoding in (
        "utf-8-sig",
        "utf-8",
        "gb18030"
    ):
        try:
            text = raw.decode(encoding)
            break
        except UnicodeDecodeError:
            continue
    else:
        raise RuntimeError(
            "无法识别 SRT 编码"
        )

    items = parse_srt(text)

    if not items:
        raise RuntimeError(
            "没有解析到有效 SRT 字幕"
        )

    output_dir = (
        input_path.parent
        / f"{input_path.stem}_5track_srt"
    )

    output_dir.mkdir(
        exist_ok=True
    )

    tracks = {
        "Cn1.srt": [],
        "Cn2.srt": [],
        "En1.srt": [],
        "En2.srt": [],
        "En3.srt": [],
    }

    for item in items:

        zh1, zh2 = split_chinese(
            item["chinese"]
        )

        en1, en2, en3 = split_english(
            item["english"]
        )

        start = item["start"]
        end = item["end"]

        tracks["Cn1.srt"].append(
            (start, end, zh1)
        )

        tracks["Cn2.srt"].append(
            (start, end, zh2)
        )

        tracks["En1.srt"].append(
            (start, end, en1)
        )

        tracks["En2.srt"].append(
            (start, end, en2)
        )

        tracks["En3.srt"].append(
            (start, end, en3)
        )

    for filename, rows in tracks.items():

        write_srt(
            output_dir / filename,
            rows
        )

    print()
    print("=" * 50)
    print("五轨 SRT 生成完成")
    print("=" * 50)
    print()
    print(f"输入：{input_path}")
    print(f"字幕块：{len(items)}")
    print(f"输出目录：{output_dir}")
    print()
    print("Cn1.srt  中文第一行")
    print("Cn2.srt  中文第二行")
    print("En1.srt  英文第一行")
    print("En2.srt  英文第二行")
    print("En3.srt  英文第三行")
    print()
    print("这是纯 SRT：")
    print("没有位置")
    print("没有坐标")
    print("没有字体")
    print("没有字幕框")
    print("=" * 50)


if __name__ == "__main__":
    main()
```

``` python
class Measurer:
    """宽度测量器：有 Pillow 就量真字体，没有就用内置度量表。"""

    def __init__(self, cfg: Config):
        self.cfg = cfg
        self.mode = "estimate"
        self._pil = {}
        self._pil_ready = False
        self.off_table_fonts: set[str] = set()   # 用了与字宽表不匹配的字体

        if cfg.measure in ("auto", "metric"):
            try:
                from PIL import ImageFont  # noqa: F401
                self._pil_ready = True
            except ImportError:
                if cfg.measure == "metric":
                    raise SystemExit(
                        "指定了 --measure metric，但没有安装 Pillow。\n"
                        "    pip install pillow      （或改用 --measure estimate）"
                    )

    # ---------- 字体文件定位 ----------
    def _font_path(self, font_name: str, override: str) -> Path | None:
        if override:
            p = Path(override)
            if not p.exists():
                raise SystemExit(f"指定的字体文件不存在：{p}")
            return p
        for cand in FONT_FILES.get(font_name.strip().lower(), []):
            p = Path(cand)
            if p.exists():
                return p
        return None

    def _pil_font(self, font_name: str, override: str):
        """以 100px 载入字体并缓存；宽度按比例缩放即可（字宽是线性的）。"""
        if not self._pil_ready:
            return None
        if font_name in self._pil:
            return self._pil[font_name]
        from PIL import ImageFont

        path = self._font_path(font_name, override)
        font = None
        if path:
            try:
                font = ImageFont.truetype(str(path), 100)
            except Exception:
                font = None
        self._pil[font_name] = (font, path)
        return self._pil[font_name]

    # ---------- 对外接口 ----------
    def width(self, text: str, size: int, kind: str) -> float:
        """返回文本在 ASS 里的近似像素宽度。kind: 'zh' | 'en'"""
        if not text:
            return 0.0
        font_name = self.cfg.zh_font if kind == "zh" else self.cfg.en_font
        override = self.cfg.zh_font_file if kind == "zh" else self.cfg.en_font_file

        got = self._pil_font(font_name, override) if self._pil_ready else None
        if got and got[0] is not None:
            font, path = got
            self.mode = "metric"
            try:
                return font.getlength(text) * size / 100.0
            except Exception:
                pass

        table = ADV["yahei"] if kind == "zh" else ADV["arial"]
        # 内置字宽表只对这两款字体是准的；换了字体还按旧表算就是静默出错
        measured_font = "microsoft yahei" if kind == "zh" else "arial"
        if font_name.strip().lower() != measured_font:
            self.off_table_fonts.add(font_name)
        total = 0
        for ch in text:
            o = ord(ch)
            if o == 0x20:
                total += table.get(" ", 280)
            elif o >= _FULLWIDTH_START:
                total += 1000
            elif ch in table:
                total += table[ch]
            else:
                total += 560
        return total * size / 1000.0

def esc(text: str) -> str:
    return (
        text.replace("\\", r"\\")
            .replace("{", r"\{")
            .replace("}", r"\}")
    )


def clean_text(text: str) -> str:
    """去掉 HTML 标签、已有 ASS 覆盖标签、零宽字符，并压缩空白。"""
    text = re.sub(r"<[^>]*>", "", text)          # <i> 之类
    text = re.sub(r"\{[^}]*\}", "", text)        # {\an8} 之类，否则会被原样显示
    text = text.replace("\u200b", "").replace("\ufeff", "")
    return " ".join(text.split())


def has_cjk(s: str) -> bool:
    return bool(_CJK_RE.search(s))


def has_latin(s: str) -> bool:
    return bool(_LATIN_RE.search(s))


# 断点禁则：上一行不能以这些词结尾
BAD_TAIL = {
    "a", "an", "the", "to", "of", "in", "on", "at", "by", "for", "with",
    "and", "or", "but", "so", "that", "this", "these", "those", "my",
    "your", "his", "her", "its", "our", "their", "from", "into", "than",
    "as", "is", "are", "was", "were", "be", "been", "am", "do", "does",
    "did", "not", "no", "very", "more", "most", "will", "would", "can",
    "could", "should", "it", "we", "they", "he", "she", "i",
}
# 以这些词开头的下一行更自然（只放并列/从属连词。
# 不要把 to / it / I / we 之类放进来：它们本身是上一行的合法结尾，
# 放在这里会让"yes to creating"被硬断成两行，反而是错的）
GOOD_HEAD = {"and", "or", "but", "so", "because", "while", "when", "then", "yet"}

# 中文可断处（断在这些标点之后更自然）
ZH_BREAK_AFTER = "，。、；：！？…—）》」』”‘’"
ZH_NO_HEAD = "，。、；：！？…—）》」』”’"


def _tokens_width(words: list[str], width_of) -> float:
    return width_of(" ".join(words))


def best_break(text: str, width_of, min_side: int = 0) -> tuple[str, str] | None:
    r"""
    找一处最自然的断点。返回 (上行, 下行) 或 None。

    打分 = 断点等级 * 0.5 + 左右平衡度(0~1)。
    等级：
        3 = 断在句末/逗号等标点之后
        2 = 断在并列/从属连词之前
        1 = 断在普通空格
        0 = 上行以介词/冠词结尾（尽量避免）
    等级只值 0.5 分，平衡度值 1 分 —— 这样"断在 and 前但上下严重不等长"
    不会压过"断得均衡但只是普通空格"，v3.2 那几个歪掉的断句就是这么来的。
    """
    words = text.split()
    if len(words) < 2:
        return None

    best = None
    for i in range(1, len(words)):
        up = words[:i]
        down = words[i:]
        if not up or not down:
            continue

        up_text = " ".join(up)
        down_text = " ".join(down)
        if min_side and (len(up_text) < min_side or len(down_text) < min_side):
            continue

        tail = up[-1].strip('"\'()[]').lower().rstrip(",.;:!?")
        head = down[0].strip('"\'()[]').lower().rstrip(",.;:!?")

        if re.search(r"[.!?…:;,、，。；：]$", up[-1]):
            cls = 3
        elif head in GOOD_HEAD:
            cls = 2
        elif tail in BAD_TAIL:
            cls = 0
        else:
            cls = 1

        wa = width_of(up_text)
        wb = width_of(down_text)
        ratio = abs(wa - wb) / max(1.0, wa + wb)
        score = cls * 0.5 + (1.0 - ratio)

        if best is None or score > best[0]:
            best = (score, up_text, down_text)

    return (best[1], best[2]) if best else None


def best_break_zh(text: str) -> tuple[str, str] | None:
    """中文没有空格，按标点优先、其次取最接近中间的位置断行。"""
    n = len(text)
    if n < 6:
        return None

    mid = n / 2
    punct = [i + 1 for i, ch in enumerate(text) if ch in ZH_BREAK_AFTER]
    cand = punct if punct else list(range(3, n - 2))
    if not cand:
        return None

    cand = [i for i in cand if 2 <= i <= n - 2]
    if not cand:
        return None

    # 标点断点优先；同为标点时取最靠近中间的
    best_i = min(
        cand,
        key=lambda i: (0 if text[i - 1] in ZH_BREAK_AFTER else 1, abs(i - mid)),
    )
    up, down = text[:best_i], text[best_i:]
    if down and down[0] in ZH_NO_HEAD:
        return None
    return (up, down) if up and down else None


def split_english(text: str, width_of, min_split_width: float = 0.0) -> tuple[str, str]:
    """英文按设计拆成上下两层，选最自然的断点。"""
    text = clean_text(text)
    if not text:
        return "", ""
    if min_split_width and width_of(text) < min_split_width:
        return text, ""
    got = best_break(text, width_of)
    if got:
        return got
    return text, ""

TS_RE = re.compile(r"(\d{1,3}):(\d{2}):(\d{2})[,.](\d{1,3})")


def parse_ts(s: str) -> int | None:
    """时间码 -> 毫秒。毫秒位数不足自动补零（v3.2 在 .44 这种写法上会算错 10 倍）。"""
    m = TS_RE.search(s)
    if not m:
        return None
    h, mi, se, ms = m.groups()
    ms = (ms + "000")[:3]
    return ((int(h) * 60 + int(mi)) * 60 + int(se)) * 1000 + int(ms)


def fmt_ass(cs: int) -> str:
    """厘秒 -> ASS 时间。"""
    h, rem = divmod(max(0, cs), 360000)
    m, rem = divmod(rem, 6000)
    s, c = divmod(rem, 100)
    return f"{h}:{m:02d}:{s:02d}.{c:02d}"


def ms_to_cs_start(ms: int) -> int:
    """开始时间向上取整到厘秒：不会把字幕提前。"""
    return -(-ms // 10)


def ms_to_cs_end(ms: int) -> int:
    """结束时间向下取整到厘秒：不会把字幕拖后。"""
    return ms // 10

@dataclass
class RawCue:
    start_ms: int
    end_ms: int
    en: str
    zh: str


@dataclass
class Skipped:
    reason: str
    raw: str


def split_bilingual(lines: list[str]) -> tuple[str, str]:
    """
    把一段文本行分成英文与中文。

    v3.2 按行号硬认「第 1 行英文、第 2 行中文」，第 3 行起静默丢弃，
    中文折两行就会丢半句。这里改为按「是否含中日韩字符」分组，
    两边各自按原始顺序拼接，纯符号行归到上一行所属语种。
    """
    en_parts: list[str] = []
    zh_parts: list[str] = []
    last = "en"

    for line in lines:
        if has_cjk(line):
            zh_parts.append(line)
            last = "zh"
        elif has_latin(line):
            en_parts.append(line)
            last = "en"
        else:
            (zh_parts if last == "zh" else en_parts).append(line)

    en = clean_text(" ".join(en_parts))
    zh = clean_text("".join(zh_parts))       # 中文拼接不加空格
    return en, zh


def parse_srt(text: str) -> tuple[list[RawCue], list[Skipped]]:
    cues: list[RawCue] = []
    skipped: list[Skipped] = []

    text = text.lstrip("\ufeff")
    blocks = re.split(r"\r?\n[ \t]*\r?\n", text.strip())

    for block in blocks:
        block = block.strip()
        if not block:
            continue

        lines = [ln.strip() for ln in block.splitlines()]
        time_idx = None
        start_ms = end_ms = None
        for i, ln in enumerate(lines):
            if "-->" not in ln:
                continue
            left, _, right = ln.partition("-->")
            s, e = parse_ts(left), parse_ts(right)
            if s is not None and e is not None:
                time_idx, start_ms, end_ms = i, s, e
                break

        if time_idx is None:
            skipped.append(Skipped("找不到时间码", block[:60]))
            continue

        body = [clean_text(ln) for ln in lines[time_idx + 1:]]
        body = [b for b in body if b]
        if not body:
            skipped.append(Skipped("没有文本", block[:60]))
            continue

        en, zh = split_bilingual(body)
        if not en and not zh:
            skipped.append(Skipped("文本为空", block[:60]))
            continue

        if end_ms <= start_ms:
            skipped.append(Skipped("结束时间不晚于开始时间", block[:60]))
            continue

        cues.append(RawCue(start_ms, end_ms, en, zh))

    return cues, skipped


def normalize_times(cues: list[RawCue], allow_overlap: bool):
    """
    转成厘秒并规整时间。

    开始向上取整、结束向下取整，这样 SRT 里
    「6,560 结束 / 6,561 开始」这种刻意的 1ms 间隔不会被抹平。
    """
    cues.sort(key=lambda c: c.start_ms)
    out = []
    clamped = 0

    for c in cues:
        start = ms_to_cs_start(c.start_ms)
        end = ms_to_cs_end(c.end_ms)
        if end <= start:
            end = start + 1
        out.append((c, start, end))

    if not allow_overlap:
        for i in range(len(out) - 1):
            cue, start, end = out[i]
            nxt_start = out[i + 1][1]
            if end > nxt_start:
                new_end = max(start + 1, nxt_start - 1)
                if new_end != end:
                    out[i] = (cue, start, new_end)
                    clamped += 1

    return out, clamped

@dataclass
class Tier:
    lines: tuple[str, ...] = ()
    hscale: float = 100.0     # \fscx/\fscy 百分比
    width: float = 0.0        # 生效后的文字宽度(px)

    @property
    def empty(self) -> bool:
        return not self.lines


@dataclass
class Slot:
    y: int          # 文字/框的中心 y
    height: int     # 框高


@dataclass
class CueOut:
    """一句字幕的最终成品：时间 + 三层排版。"""
    start_cs: int
    end_cs: int
    zh: Tier
    en1: Tier
    en2: Tier


@dataclass
class Layout:
    zh: Slot
    en1: Slot
    en2: Slot
    check: list[str] = field(default_factory=list)


def fit_tier(
    text: str,
    size: int,
    kind: str,
    cfg: Config,
    meas: Measurer,
    max_text_w: float,
    allow_reflow: bool = True,
) -> Tier:
    """
    让一层文字一定放得下：
      放得下 -> 一行
      放不下 -> 先按语义折成两行
      还放不下 -> 等比缩小 (\fscx\fscy)
    """
    text = text.strip()
    if not text:
        return Tier()

    width_of = lambda t: meas.width(t, size, kind)
    w = width_of(text)
    if w <= max_text_w:
        return Tier((text,), 100.0, w)

    if allow_reflow:
        br = best_break_zh(text) if kind == "zh" else best_break(text, width_of)
        if br:
            a, b = br
            wa, wb = width_of(a), width_of(b)
            if max(wa, wb) <= max_text_w:
                return Tier((a, b), 100.0, max(wa, wb))

    k = max(1.0, max_text_w / w) * 100
    return Tier((text,), round(k, 1), max_text_w)


def compute_layout(cfg: Config, max_lines_zh: int, max_lines_en: int, video_h: int) -> Layout:
    """
    版式自下而上堆叠。

    v3.2 是写死 EN1_Y=975 / EN2_Y=1025，中心距 50px 小于框高 64px，
    两个半透明框必然叠画（实测重叠 14px，重叠处颜色更深）。
    这里改成：先量出两层英文各自需要多高，再从画面底部往上排，
    中间留 en_gap，所以无论文字一行还是两行都不可能重叠。

    注意：槽位用的是全片最大行数，因此每一句字幕的框位置都一样，
    不会出现"这句低一点那句高一点"的跳动。
    """
    check: list[str] = []

    zh_step = int(round(cfg.zh_size * 1.35))
    en_step = int(round(cfg.en_size * 1.20))

    h_zh = cfg.box_height_zh + max(0, max_lines_zh - 1) * zh_step
    h_en1 = cfg.box_height_en + max(0, max_lines_en - 1) * en_step
    h_en2 = cfg.box_height_en + max(0, max_lines_en - 1) * en_step

    # ---------- 中文：按中心定位（与 v3.2 的 ZH_Y 语义一致） ----------
    zh_y = cfg.zh_y
    zh_top = zh_y - h_zh // 2
    zh_bottom = zh_top + h_zh

    # ---------- 英文：底部往上 ----------
    en2_bottom = video_h - cfg.en_bottom_margin
    en2_y = en2_bottom - h_en2 // 2
    en2_top = en2_bottom - h_en2

    en1_bottom = en2_top - cfg.en_gap
    en1_y = en1_bottom - h_en1 // 2
    en1_top = en1_bottom - h_en1

    if zh_top < 0:
        check.append(
            f"中文框上边缘 {zh_top} 已出画（改小 --zh-y 或中文折行过多），"
            f"建议 --zh-y 不小于 {h_zh // 2}"
        )
    if en1_top < zh_bottom:
        check.append(
            f"版式过挤：中文框底边 {zh_bottom} 已越过英文框顶边 {en1_top}，"
            f"建议减小字号或调整 --zh-y / --en-bottom-margin"
        )
    if en1_bottom - en1_top != h_en1 or en2_bottom - en2_top != h_en2:
        check.append("内部错误：槽位高度计算不一致")

    return Layout(
        zh=Slot(zh_y, h_zh),
        en1=Slot(en1_y, h_en1),
        en2=Slot(en2_y, h_en2),
        check=check,
    )

def ass_color(hex_color: str, transparency: int = 0) -> str:
    """#RRGGBB -> &HAABBGGRR（AA: 00 不透明, FF 全透明）"""
    c = hex_color.lstrip("#")
    if len(c) == 3:
        c = "".join(ch * 2 for ch in c)
    r, g, b = c[0:2], c[2:4], c[4:6]
    a = round(max(0, min(100, transparency)) * 255 / 100)
    return f"&H{a:02X}{b}{g}{r}".upper()

    return fmt_ass(cs)

def dialogue_text(
    layer: int, start: str, end: str, style: str, tier: Tier, x: int, y: int, fade: int
) -> str:
    body = r"\N".join(esc(ln) for ln in tier.lines)
    tag = [f"\\pos({x},{y})"]
    if abs(tier.hscale - 100.0) > 0.05:
        tag.append(f"\\fscx{tier.hscale:g}\\fscy{tier.hscale:g}")
    if fade > 0:
        tag.append(f"\\fad({fade},{fade})")
    return (
        f"Dialogue: {layer},{start},{end},{style},,0,0,0,,"
        f"{{{''.join(tag)}}}{body}"
    )


def dialogue_box(
    layer: int, start: str, end: str, style: str,
    width: float, height: float, x: float, y: float, radius: int, fade: int
) -> str:
    """用 ASS Drawing Mode 画圆角矩形。b 为三次贝塞尔，0.5522848 是 90° 圆弧控制点系数。"""
    width = max(1, round(width))
    height = max(1, round(height))
    r = max(0, min(int(radius), width // 2, height // 2))

    left = round(x - width / 2)
    top = round(y - height / 2)
    right = round(x + width / 2)
    bottom = round(y + height / 2)

    if r == 0:
        drawing = (
            f"m {left} {top} l {right} {top} l {right} {bottom} "
            f"l {left} {bottom} l {left} {top}"
        )
    else:
        c = round(r * 0.5522848)
        drawing = (
            f"m {left + r} {top} "
            f"l {right - r} {top} "
            f"b {right - r + c} {top} {right} {top + r - c} {right} {top + r} "
            f"l {right} {bottom - r} "
            f"b {right} {bottom - r + c} {right - r + c} {bottom} {right - r} {bottom} "
            f"l {left + r} {bottom} "
            f"b {left + r - c} {bottom} {left} {bottom - r + c} {left} {bottom - r} "
            f"l {left} {top + r} "
            f"b {left} {top + r - c} {left + r - c} {top} {left + r} {top}"
        )

    pre = "\\p1" + (f"\\fad({fade},{fade})" if fade > 0 else "")
    return (
        f"Dialogue: {layer},{start},{end},{style},,0,0,0,,"
        f"{{{pre}}}{drawing}{{\\p0}}"
    )


@dataclass
class Built:
    lines: list[str] = field(default_factory=list)
    stats: dict = field(default_factory=dict)


def build_ass(
    cues: "list[CueOut]",
    layout: Layout,
    cfg: Config,
    video_w: int,
    video_h: int,
    duration_s: float | None = None,
) -> Built:
    max_box_w = video_w - 2 * cfg.margin_x

    out: list[str] = [build_header(cfg, video_w, video_h)]
    cx = video_w // 2

    # 遮盖带：一条贯穿全片的绘制事件，压在所有框/文字下面（layer 0）。
    # 结束时间用视频真实时长；拿不到才退回 10 小时占位（避免 Aegisub 里出现超长事件）。
    if cfg.cover_top > 0:
        band_h = video_h - cfg.cover_top
        if duration_s and duration_s > 0:
            band_end = fmt_ass(int(duration_s * 100) + 1)
        else:
            band_end = "9:59:59.99"
        out.append(
            dialogue_box(
                0, "0:00:00.00", band_end, "CoverBand",
                video_w, band_h, video_w / 2, cfg.cover_top + band_h / 2,
                0, 0,
            )
        )

    max_used_w = 0.0
    scaled_tiers = 0
    reflowed_tiers = 0

    for cue in cues:
        s, e = ass_time(cue.start_cs), ass_time(cue.end_cs)

        for tier, slot, style, box_style in (
            (cue.zh, layout.zh, "Chinese", "BoxChinese"),
            (cue.en1, layout.en1, "English1", "BoxEnglish1"),
            (cue.en2, layout.en2, "English2", "BoxEnglish2"),
        ):
            if tier.empty:
                continue

            if abs(tier.hscale - 100.0) > 0.05:
                scaled_tiers += 1
            if len(tier.lines) > 1:
                reflowed_tiers += 1

            box_w = min(max_box_w, max(cfg.box_min_width, tier.width + 2 * cfg.box_padding_x))
            max_used_w = max(max_used_w, box_w)

            out.append(
                dialogue_box(
                    1, s, e, box_style, box_w, slot.height, cx, slot.y,
                    cfg.box_radius, cfg.fade,
                )
            )
            out.append(
                dialogue_text(2, s, e, style, tier, cx, slot.y, cfg.fade)
            )

    stats = {
        "max_box_width": max_used_w,
        "max_box_limit": max_box_w,
        "scaled_tiers": scaled_tiers,
        "reflowed_tiers": reflowed_tiers,
    }
    return Built(out, stats)


def prepare_cues(
    timed: list[tuple], cfg: Config, meas: Measurer, max_text_w: float
) -> tuple[list["CueOut"], dict]:
    """把已经归整过时间的字幕变成各层的最终排版（含英文两层拆分）。"""
    prepared: list[CueOut] = []
    max_en_lines = 1
    max_zh_lines = 1
    split_ok = 0

    for cue, start_cs, end_cs in timed:
        en1_text, en2_text = "", ""
        if cue.en:
            if cfg.split_english:
                en1_text, en2_text = split_english(
                    cue.en,
                    lambda t: meas.width(t, cfg.en_size, "en"),
                    cfg.min_split_width,
                )
                if en2_text:
                    split_ok += 1
            else:
                en1_text = clean_text(cue.en)

        zh = fit_tier(cue.zh, cfg.zh_size, "zh", cfg, meas, max_text_w)
        en1 = fit_tier(en1_text, cfg.en_size, "en", cfg, meas, max_text_w)
        en2 = fit_tier(en2_text, cfg.en_size, "en", cfg, meas, max_text_w)

        max_zh_lines = max(max_zh_lines, len(zh.lines))
        max_en_lines = max(max_en_lines, len(en1.lines), len(en2.lines))

        prepared.append(CueOut(start_cs, end_cs, zh, en1, en2))

    return prepared, {
        "split_ok": split_ok,
        "max_lines_zh": max_zh_lines,
        "max_lines_en": max_en_lines,
    }

VIDEO_EXT = {".mp4", ".mov", ".mkv", ".avi", ".m4v", ".webm"}


def ffprobe_size(video: Path) -> tuple[int, int] | None:
    exe = shutil.which("ffprobe")
    if not exe:
        return None
    cmd = [
        exe, "-v", "error", "-select_streams", "v:0",
        "-show_entries", "stream=width,height", "-of", "csv=p=0",
        str(video),
    ]
    try:
        r = subprocess.run(cmd, capture_output=True, text=True, timeout=30)
        w, h = r.stdout.strip().splitlines()[0].split(",")[:2]
        return int(w), int(h)
    except Exception:
        return None


def ffprobe_duration(video: Path) -> float | None:
    """视频真实时长（秒）。拿不到就返回 None。"""
    exe = shutil.which("ffprobe")
    if not exe:
        return None
    cmd = [exe, "-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", str(video)]
    try:
        r = subprocess.run(cmd, capture_output=True, text=True, timeout=30)
        return float(r.stdout.strip().splitlines()[0])
    except Exception:
        return None


def escape_filter_path(p: Path) -> str:
    """ffmpeg 滤镜里路径要转义，Windows 上冒号和反斜杠是重灾区。"""
    s = str(p).replace("\\", "/")
    s = s.replace(":", "\\:")
    return f"'{s}'"


def burn_video(video: Path, ass: Path, out: Path, crf: int) -> bool:
    exe = shutil.which("ffmpeg")
    if not exe:
        print("  [!] 没找到 ffmpeg，跳过烧制")
        return False
    cmd = [
        exe, "-y", "-i", str(video),
        "-vf", f"ass={escape_filter_path(ass)}",
        "-c:v", "libx264", "-crf", str(crf), "-preset", "slow",
        "-c:a", "copy", str(out),
    ]
    print("  烧制中：" + " ".join(cmd))
    r = subprocess.run(cmd)
    return r.returncode == 0

def read_srt(path: Path) -> str:
    raw = path.read_bytes()
    for enc in ("utf-8-sig", "utf-8", "gb18030", "cp1252"):
        try:
            return raw.decode(enc)
        except UnicodeDecodeError:
            continue
    raise SystemExit("无法识别 SRT 编码，请另存为 UTF-8。")

def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)

    input_path = Path(args.input)
    if not input_path.exists():
        print(f"找不到文件：{input_path}")
        return 1

    output_path = (
        Path(args.output) if args.output
        else input_path.with_name(input_path.stem + "_final.ass")
    )

    cfg = Config(
        zh_font=args.zh_font, en_font=args.en_font,
        zh_size=args.zh_size, en_size=args.en_size,
        zh_color=args.zh_color, en1_color=args.en1_color, en2_color=args.en2_color,
        box_color=args.box_color, box_alpha=args.box_alpha,
        outline_color=args.outline_color,
        box_padding_x=args.box_padding_x, box_min_width=args.box_min_width,
        box_height_zh=args.box_height_zh, box_height_en=args.box_height_en,
        box_radius=args.box_radius,
        zh_y=args.zh_y, en_bottom_margin=args.en_bottom_margin,
        en_gap=args.en_gap, margin_x=args.margin_x,
        split_english=args.split_english, min_split_width=args.min_split_width,
        fade=args.fade, allow_overlap=args.allow_overlap, bom=args.bom,
        measure=args.measure,
        zh_font_file=args.zh_font_file, en_font_file=args.en_font_file,
        cover_top=args.cover_band, cover_color=args.cover_color,
    )

    # ---------- 分辨率 ----------
    video_path = Path(args.video) if args.video else None
    if video_path and not video_path.exists():
        print(f"找不到视频：{video_path}")
        return 1
    if video_path is None:
        siblings = [p for p in input_path.parent.iterdir()
                    if p.suffix.lower() in VIDEO_EXT and p.is_file()]
        if len(siblings) == 1:
            video_path = siblings[0]
            if not args.quiet:
                print(f"自动识别到同目录视频：{video_path.name}")

    if args.width and args.height:
        video_w, video_h = args.width, args.height
        src = "命令行指定"
    elif video_path:
        got = ffprobe_size(video_path)
        if got:
            video_w, video_h = got
            src = f"ffprobe {video_path.name}"
        else:
            video_w, video_h = REF_W, REF_H
            src = "ffprobe 不可用，回退 1920x1080"
    else:
        video_w, video_h = REF_W, REF_H
        src = "默认 1920x1080"

    scale = min(video_w / REF_W, video_h / REF_H)
    cfg = cfg.scaled(scale)

    # ---------- 参数校验 ----------
    if cfg.cover_top:
        if cfg.cover_top < 0 or cfg.cover_top >= video_h - 20:
            print(f"[!] --cover-band 取值无效：{args.cover_band}"
                  f"（需在 0 ~ {video_h - 20} 之间，0=关闭）")
            return 1

    # ---------- 解析 ----------
    text = read_srt(input_path)
    raw_cues, skipped = parse_srt(text)
    if not raw_cues:
        print("没有解析到有效字幕。")
        for s in skipped[:5]:
            print(f"  - {s.reason}: {s.raw}")
        return 1

    timed, clamped = normalize_times(raw_cues, cfg.allow_overlap)

    meas = Measurer(cfg)
    max_box_w = video_w - 2 * cfg.margin_x
    max_text_w = max_box_w - 2 * cfg.box_padding_x

    prepared, prep_stats = prepare_cues(timed, cfg, meas, max_text_w)
    layout = compute_layout(cfg, prep_stats["max_lines_zh"], prep_stats["max_lines_en"], video_h)

    duration = ffprobe_duration(video_path) if video_path else None

    built = build_ass(prepared, layout, cfg, video_w, video_h, duration)

    # ---------- 输出 ----------
    output_path.parent.mkdir(parents=True, exist_ok=True)
    encoding = "utf-8-sig" if cfg.bom else "utf-8"
    output_path.write_text("\n".join(built.lines), encoding=encoding)

    report(
        args, cfg, video_w, video_h, src, input_path, output_path,
        raw_cues, skipped, clamped, layout, prep_stats, built.stats,
        prepared, meas,
    )

    if args.burn:
        burn_video(video_path or input_path, output_path, Path(args.burn), args.crf)

    return 0


def report(
    args, cfg, video_w, video_h, src, input_path, output_path,
    raw_cues, skipped, clamped, layout, prep_stats, stats, prepared, meas,
) -> None:
    if args.quiet:
        return

    total_gap = None
    gaps = [b.start_cs - a.end_cs for a, b in zip(prepared, prepared[1:])]
    if gaps:
        total_gap = min(gaps)

    print()
    print("=" * 56)
    print(f"三层字幕 v{VERSION} 生成完成")
    print("=" * 56)
    print(f"输入      : {input_path}")
    print(f"输出      : {output_path}")
    print(f"画布      : {video_w}x{video_h}（{src}，缩放系数 {video_w / REF_W:.3f}）")
    if meas.mode == "metric":
        print("宽度测量  : Pillow 逐字精确测量（含字间距）")
    else:
        print("宽度测量  : 内置实测字宽表（Arial Bold / 微软雅黑 Bold 逐字实测，")
        print("            不含字偶间距，误差约 1%~2%；装 Pillow 后自动升级为精确测量）")
        if meas.off_table_fonts:
            names = "、".join(sorted(meas.off_table_fonts))
            print(f"  [!] 换了字体但字宽表只有雅黑/Arial 的度量：{names}")
            print("      框宽会按旧表估算，可能不准——装 Pillow 或用 --measure metric 可强制精确测量")
    print(f"字幕条数  : {len(raw_cues)}")
    print(f"跳过条数  : {len(skipped)}")
    for s in skipped[:5]:
        print(f"            - {s.reason}: {s.raw}")
    if len(skipped) > 5:
        print(f"            ... 另有 {len(skipped) - 5} 条")
    print(f"时间规整  : 因重叠被截短 {clamped} 条"
          + (f"；最小间隔 {total_gap} 厘秒" if total_gap is not None else ""))
    print(f"英文拆层  : {prep_stats['split_ok']} / {len(raw_cues)} 条拆成上下两层")
    print()
    print("版式（自下而上堆叠，结构上不会重叠）:")
    print(f"  中文层  y={layout.zh.y:<5} 框高={layout.zh.height}")
    print(f"  英文1层 y={layout.en1.y:<5} 框高={layout.en1.height}"
          f"   底边={layout.en1.y + layout.en1.height // 2}")
    print(f"  英文2层 y={layout.en2.y:<5} 框高={layout.en2.height}"
          f"   顶边={layout.en2.y - layout.en2.height // 2}")
    print(f"  英文两层缝隙 = {layout.en2.y - layout.en2.height // 2 - (layout.en1.y + layout.en1.height // 2)}px"
          f"（设定 {cfg.en_gap}px，负值即为叠画）")
    if cfg.cover_top > 0:
        en1_top = layout.en1.y - layout.en1.height // 2
        print(f"  遮盖带  y={cfg.cover_top}→{video_h}  颜色 {cfg.cover_color}"
              f"（盖住烧死字幕；英文两层应完全落在带内）")
        if en1_top < cfg.cover_top:
            print(f"  [!] 英文1层顶边 {en1_top} 高于遮盖带顶边 {cfg.cover_top}，"
                  f"会有半个框露在色带外面")
    print()
    print("自检:")
    print(f"  最宽背景框 {stats['max_box_width']:.0f}px / 上限 {stats['max_box_limit']}px"
          + ("  [OK]" if stats["max_box_width"] <= stats["max_box_limit"] + 0.5 else "  [超宽!]"))
    print(f"  折行层数   {stats['reflowed_tiers']}")
    print(f"  缩小层数   {stats['scaled_tiers']}")
    for msg in layout.check:
        print(f"  [!] {msg}")

    if args.dump:
        print()
        print(f"逐句几何（共 {len(prepared)} 条，显示前 10 条）:")
        for cue in prepared[:10]:
            print(f"  {fmt_ass(cue.start_cs)} -> {fmt_ass(cue.end_cs)}")
            for label, t in (("zh", cue.zh), ("en1", cue.en1), ("en2", cue.en2)):
                if t.empty:
                    continue
                print(f"      {label:<4} w={t.width:6.0f} scale={t.hscale:5.1f}%  {t.lines}")
    print("=" * 56)


if __name__ == "__main__":
    sys.exit(main())
```

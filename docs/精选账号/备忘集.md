# 我现在害怕和朋友一起旅行，因为上次我做这个视频时
#
# 输出：
#
#             中文主体
#       English 第一部分（淡）
#       English 第二部分（深）
#
# 注意：
# 英文不是从 SRT 中找第二条字幕。
# 而是把当前字幕块中的英文自动拆成上下两部分。
# ============================================================

VIDEO_WIDTH = 1920
VIDEO_HEIGHT = 1080

# ---------- 字体 ----------
ZH_FONT = "Microsoft YaHei"
EN_FONT = "Arial"

ZH_SIZE = 32
EN_SIZE = 46

# ---------- 颜色 ----------
ZH_COLOR = "#B8B8B8"
EN1_COLOR = "#B8B8B8"       # 英文上半句：淡
EN2_COLOR = "#FFFFFF"       # 英文下半句：深/亮

# ---------- 字幕框 ----------
BOX_COLOR = "#111111"
BOX_ALPHA = 35
OUTLINE_COLOR = "#000000"

# ---------- 三层位置 ----------
# 使用绝对位置，不使用 MarginV 猜位置。
ZH_Y = 50
EN1_Y = 975
EN2_Y = 1025

MARGIN_L = 100
MARGIN_R = 100


def ass_color(hex_color, transparency=0):
    c = hex_color.lstrip("#")
    r, g, b = c[0:2], c[2:4], c[4:6]
    alpha = round(max(0, min(100, transparency)) * 255 / 100)
    return f"&H{alpha:02X}{b}{g}{r}".upper()


def ass_time(t):
    t = t.replace(",", ".")
    h, m, rest = t.split(":")
    s, ms = rest.split(".")
    cs = round(int(ms) / 10)
    if cs >= 100:
        cs = 0
        s = str(int(s) + 1)
    return f"{int(h)}:{int(m):02d}:{int(s):02d}.{cs:02d}"


def clean_text(text):
    text = re.sub(r"<[^>]+>", "", text)
    text = text.replace("\u200b", "")
    return " ".join(text.split())


def split_english(text):
    """
    把一个 SRT 英文字幕拆成上下两部分。

    优先级：
    1. 如果原文有两个自然句子，在句号/问号/感叹号等处分开。
    2. 如果只有一个句子，则按接近中间的位置，从单词边界拆开。
    3. 不在单词中间硬切。
    """

    text = clean_text(text)

    if not text:
        return "", ""

    # 优先找自然句子边界。
    # 只在确实能形成两部分时使用。
    sentence_parts = re.split(r"(?<=[.!?])\s+", text)

    if len(sentence_parts) >= 2:
        # 尽量让上下两部分长度接近
        best = None
        best_diff = None

        for i in range(1, len(sentence_parts)):
            a = " ".join(sentence_parts[:i]).strip()
            b = " ".join(sentence_parts[i:]).strip()

            if not a or not b:
                continue

            diff = abs(len(a) - len(b))

            if best_diff is None or diff < best_diff:
                best = (a, b)
                best_diff = diff

        if best:
            return best

    # 没有自然句号：按单词边界拆成接近两半
    words = text.split()

    if len(words) <= 1:
        return text, ""

    # 以字符长度为目标，而不是简单按单词数量一刀切
    total = len(text)
    target = total / 2

    current = 0
    split_at = 1
    best_distance = float("inf")

    for i in range(1, len(words)):
        current += len(words[i - 1]) + 1
        distance = abs(current - target)

        if distance < best_distance:
            best_distance = distance
            split_at = i

    first = " ".join(words[:split_at]).strip()
    second = " ".join(words[split_at:]).strip()

    return first, second


def parse_srt(text):
    """
    解析实际字幕结构：
      第一个文本行 = 英文
      第二个文本行 = 中文

    时间轴保持完全一致。
    """
    result = []

    blocks = re.split(r"\r?\n\s*\r?\n", text.strip())

    for block in blocks:
        lines = [
            clean_text(x.strip("\ufeff"))
            for x in block.splitlines()
            if x.strip()
        ]

        if len(lines) < 3:
            continue

        # 找时间轴
        time_index = None
        match = None

        for i, line in enumerate(lines):
            m = re.match(
                r"(\d\d:\d\d:\d\d[,.]\d{3})\s*-->\s*"
                r"(\d\d:\d\d:\d\d[,.]\d{3})",
                line
            )
            if m:
                time_index = i
                match = m
                break

        if not match:
            continue

        subtitle_lines = lines[time_index + 1:]

        if len(subtitle_lines) < 2:
            continue

        english = subtitle_lines[0]
        chinese = subtitle_lines[1]

        en1, en2 = split_english(english)

        result.append({
            "start": ass_time(match.group(1)),
            "end": ass_time(match.group(2)),
            "zh": chinese,
            "en1": en1,
            "en2": en2,
        })

    return result


def esc(text):
    return (
        text.replace("\\", r"\\")
            .replace("{", r"\{")
            .replace("}", r"\}")
    )


def dialogue(layer, start, end, style, text, y):
    if not text:
        return None

    x = VIDEO_WIDTH // 2

    return (
        f"Dialogue: {layer},{start},{end},{style},"
        f",0,0,0,,{{\\pos({x},{y})}}{esc(text)}"
    )


def make_ass(items, output):
    white = ass_color(ZH_COLOR)
    light = ass_color(EN1_COLOR)
    bright = ass_color(EN2_COLOR)
    outline = ass_color(OUTLINE_COLOR)
    box = ass_color(BOX_COLOR, BOX_ALPHA)

    header = f"""[Script Info]
ScriptType: v4.00+
PlayResX: {VIDEO_WIDTH}
PlayResY: {VIDEO_HEIGHT}
WrapStyle: 2
ScaledBorderAndShadow: yes

[V4+ Styles]
Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding
Style: Chinese,{ZH_FONT},{ZH_SIZE},{white},{white},{outline},{box},-1,0,0,0,100,100,0,0,3,1,0,5,{MARGIN_L},{MARGIN_R},0,1
Style: English1,{EN_FONT},{EN_SIZE},{light},{light},{outline},{box},-1,0,0,0,100,100,0,0,3,1,0,5,{MARGIN_L},{MARGIN_R},0,1
Style: English2,{EN_FONT},{EN_SIZE},{bright},{bright},{outline},{box},-1,0,0,0,100,100,0,0,3,1,0,5,{MARGIN_L},{MARGIN_R},0,1

[Events]
Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text
"""

    lines = [header]

    for item in items:
        start = item["start"]
        end = item["end"]

        # 中文主体
        line = dialogue(0, start, end, "Chinese", item["zh"], ZH_Y)
        if line:
            lines.append(line)

        # 英文第一部分：淡
        line = dialogue(1, start, end, "English1", item["en1"], EN1_Y)
        if line:
            lines.append(line)

        # 英文第二部分：亮
        line = dialogue(2, start, end, "English2", item["en2"], EN2_Y)
        if line:
            lines.append(line)

    output.write_text("\n".join(lines), encoding="utf-8-sig")


def main():
    if len(sys.argv) < 2:
        print("用法：python make_ass_final_v3.py input.srt [output.ass]")
        raise SystemExit(1)

    input_path = Path(sys.argv[1])

    if not input_path.exists():
        print(f"找不到文件：{input_path}")
        raise SystemExit(1)

    output_path = (
        Path(sys.argv[2])
        if len(sys.argv) >= 3
        else input_path.with_name(input_path.stem + "_final.ass")
    )

    raw = input_path.read_bytes()

    for encoding in ("utf-8-sig", "utf-8", "gb18030"):
        try:
            text = raw.decode(encoding)
            break
        except UnicodeDecodeError:
            continue
    else:
        print("无法识别 SRT 编码。")
        raise SystemExit(1)

    items = parse_srt(text)

    if not items:
        print("没有解析到有效字幕。")
        raise SystemExit(1)

    make_ass(items, output_path)

    print()
    print("========================================")
    print("三层字幕 v3 生成完成")
    print("========================================")
    print(f"输入：{input_path}")
    print(f"输出：{output_path}")
    print(f"字幕块：{len(items)}")
    print()
    print("实际处理逻辑：")
    print("  原 SRT：英文 + 中文")
    print("  ↓")
    print("  英文自动拆成上下两部分")
    print("  ↓")
    print("  中文最上")
    print("  英文第一部分：较淡")
    print("  英文第二部分：较亮")
    print("========================================")


if __name__ == "__main__":
    main()

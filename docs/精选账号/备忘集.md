from pathlib import Path
import re
import sys

# ============================================================
# 三层双语字幕 v3.2
#
# 核心：
# 1. SRT 第一行英文、第二行中文
# 2. 英文自动拆成上下两部分
# 3. 中文在上，英文两层在下
# 4. 英文第一部分较淡，第二部分较亮
# 5. 真正计算文字宽度，使用 ASS 绘制独立背景矩形
# 6. BOX_PADDING_X / BOX_MIN_WIDTH 真正以像素控制
#
# 注意：
# 本版本不再使用 \h 空格“撑”字幕框。
# 字幕文字和背景框分开绘制，因此宽度控制是真实的。
# ============================================================

VIDEO_WIDTH = 1920
VIDEO_HEIGHT = 1080

# ---------- 字体 ----------
ZH_FONT = "Microsoft YaHei"
EN_FONT = "Arial"

ZH_SIZE = 32
EN_SIZE = 46

# ---------- 颜色 ----------
ZH_COLOR = "#B8B8B8"
EN1_COLOR = "#B8B8B8"
EN2_COLOR = "#FFFFFF"

# ---------- 字幕框 ----------
BOX_COLOR = "#111111"
BOX_ALPHA = 35
OUTLINE_COLOR = "#000000"

# 真正的左右像素留白
BOX_PADDING_X = 30

# 字幕框最小宽度
BOX_MIN_WIDTH = 100

# 字幕框高度
BOX_HEIGHT_ZH = 54
BOX_HEIGHT_EN = 64

# 字幕框圆角半径（像素）
# 0 = 直角；20~30 = 轻微圆角；40~50 = 明显圆角
BOX_RADIUS = 20

# ---------- 三层位置 ----------
ZH_Y = 50
EN1_Y = 975
EN2_Y = 1025

# ---------- 字体宽度估算 ----------
# Arial / Microsoft YaHei 的实际渲染宽度会因字符不同而变化。
# 这里使用经验系数作为 ASS 位置框的近似计算。
ZH_CHAR_WIDTH = 0.95
EN_CHAR_WIDTH = 0.56

MARGIN_L = 100
MARGIN_R = 100


def ass_color(hex_color, transparency=0):
    c = hex_color.lstrip("#")
    r, g, b = c[0:2], c[2:4], c[4:6]
    alpha = round(max(0, min(100, transparency)) * 255 / 100)
    return f"&H{alpha:02X}{b}{g}{r}".upper()


def ass_time(t):
    t = t.replace(",", ".")
    h, m, rest = t.split(":")
    s, ms = rest.split(".")

    total_cs = round(int(ms) / 10)
    total_seconds = int(h) * 3600 + int(m) * 60 + int(s)

    if total_cs >= 100:
        total_seconds += total_cs // 100
        total_cs %= 100

    hours = total_seconds // 3600
    minutes = (total_seconds % 3600) // 60
    seconds = total_seconds % 60

    return f"{hours}:{minutes:02d}:{seconds:02d}.{total_cs:02d}"


def clean_text(text):
    text = re.sub(r"<[^>]+>", "", text)
    text = text.replace("\u200b", "")
    return " ".join(text.split())


def split_english(text):
    text = clean_text(text)

    if not text:
        return "", ""

    sentence_parts = re.split(r"(?<=[.!?])\s+", text)

    if len(sentence_parts) >= 2:
        best = None
        best_diff = None

        for i in range(1, len(sentence_parts)):
            a = " ".join(sentence_parts[:i]).strip()
            b = " ".join(sentence_parts[i:]).strip()

            if not a or not b:
                continue

            diff = abs(len(a) - len(b))

            if best_diff is None or diff < best_diff:
                best = (a, b)
                best_diff = diff

        if best:
            return best

    words = text.split()

    if len(words) <= 1:
        return text, ""

    total = len(text)
    target = total / 2

    current = 0
    split_at = 1
    best_distance = float("inf")

    for i in range(1, len(words)):
        current += len(words[i - 1]) + 1
        distance = abs(current - target)

        if distance < best_distance:
            best_distance = distance
            split_at = i

    return (
        " ".join(words[:split_at]).strip(),
        " ".join(words[split_at:]).strip()
    )


def parse_srt(text):
    result = []

    blocks = re.split(r"\r?\n\s*\r?\n", text.strip())

    for block in blocks:
        raw_lines = [
            x.strip("\ufeff").strip()
            for x in block.splitlines()
            if x.strip()
        ]

        if len(raw_lines) < 3:
            continue

        match = None
        time_index = None

        for i, raw_line in enumerate(raw_lines):
            m = re.match(
                r"(\d\d:\d\d:\d\d[,.]\d{3})\s*-->\s*"
                r"(\d\d:\d\d:\d\d[,.]\d{3})",
                raw_line
            )
            if m:
                match = m
                time_index = i
                break

        if not match:
            continue

        subtitle_lines = [
            clean_text(line)
            for line in raw_lines[time_index + 1:]
        ]

        if len(subtitle_lines) < 2:
            continue

        english = subtitle_lines[0]
        chinese = subtitle_lines[1]

        en1, en2 = split_english(english)

        result.append({
            "start": ass_time(match.group(1)),
            "end": ass_time(match.group(2)),
            "zh": chinese,
            "en1": en1,
            "en2": en2,
        })

    return result


def esc(text):
    return (
        text.replace("\\", r"\\")
            .replace("{", r"\{")
            .replace("}", r"\}")
    )


def estimate_text_width(text, font_size, char_width):
    """
    估算字幕文字宽度。

    对英文按字符分别估算：
    - 空格更窄
    - 大写略宽
    - 标点较窄

    对中文按字符估算。
    """
    width = 0.0

    for ch in text:
        if ch == " ":
            width += font_size * 0.28
        elif ord(ch) >= 0x2E80:
            width += font_size * char_width
        elif ch in ".,!?;:'\"-()[]":
            width += font_size * 0.30
        elif ch.isupper():
            width += font_size * char_width * 1.08
        else:
            width += font_size * char_width

    return max(1, round(width))


def box_width(text, font_size, char_width):
    text_width = estimate_text_width(
        text,
        font_size,
        char_width
    )

    return max(
        BOX_MIN_WIDTH,
        text_width + BOX_PADDING_X * 2
    )


def dialogue_text(
    layer,
    start,
    end,
    style,
    text,
    x,
    y,
):
    if not text:
        return None

    return (
        f"Dialogue: {layer},{start},{end},{style},"
        f",0,0,0,,{{\\pos({x},{y})}}{esc(text)}"
    )


def dialogue_box(
    layer,
    start,
    end,
    style,
    width,
    height,
    x,
    y,
):
    """
    使用 ASS Drawing Mode 绘制圆角矩形背景。

    ASS 的 b 命令用于三次 Bézier 曲线。
    这里用 0.5522848 的控制点近似四个 90° 圆弧。
    """

    width = max(1, round(width))
    height = max(1, round(height))

    radius = max(0, min(
        round(BOX_RADIUS),
        width // 2,
        height // 2
    ))

    left = round(x - width / 2)
    top = round(y - height / 2)
    right = round(x + width / 2)
    bottom = round(y + height / 2)

    if radius == 0:
        drawing = (
            f"{{\\p1}}"
            f"m {left} {top} "
            f"l {right} {top} "
            f"l {right} {bottom} "
            f"l {left} {bottom} "
            f"l {left} {top}"
            f"{{\\p0}}"
        )
    else:
        # 三次 Bézier 圆角控制点系数
        k = 0.5522848
        c = round(radius * k)

        drawing = (
            f"{{\\p1}}"
            # 从顶部左侧圆角结束点开始
            f"m {left + radius} {top} "
            # 顶边
            f"l {right - radius} {top} "
            # 右上圆角
            f"b {right - radius + c} {top} "
            f"{right} {top + radius - c} "
            f"{right} {top + radius} "
            # 右边
            f"l {right} {bottom - radius} "
            # 右下圆角
            f"b {right} {bottom - radius + c} "
            f"{right - radius + c} {bottom} "
            f"{right - radius} {bottom} "
            # 底边
            f"l {left + radius} {bottom} "
            # 左下圆角
            f"b {left + radius - c} {bottom} "
            f"{left} {bottom - radius + c} "
            f"{left} {bottom - radius} "
            # 左边
            f"l {left} {top + radius} "
            # 左上圆角
            f"b {left} {top + radius - c} "
            f"{left + radius - c} {top} "
            f"{left + radius} {top}"
            f"{{\\p0}}"
        )

    return (
        f"Dialogue: {layer},{start},{end},{style},"
        f",0,0,0,,{drawing}"
    )


def make_ass(items, output):
    zh_color = ass_color(ZH_COLOR)
    en1_color = ass_color(EN1_COLOR)
    en2_color = ass_color(EN2_COLOR)
    outline = ass_color(OUTLINE_COLOR)
    box = ass_color(BOX_COLOR, BOX_ALPHA)

    # 背景矩形专用样式。
    # Alignment=7 只是为了让绘图层保持稳定，实际坐标由 m/l 控制。
    header = f"""[Script Info]
ScriptType: v4.00+
PlayResX: {VIDEO_WIDTH}
PlayResY: {VIDEO_HEIGHT}
WrapStyle: 2
ScaledBorderAndShadow: yes

[V4+ Styles]
Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding
Style: Chinese,{ZH_FONT},{ZH_SIZE},{zh_color},{zh_color},{outline},{box},-1,0,0,0,100,100,0,0,1,1,0,5,{MARGIN_L},{MARGIN_R},0,1
Style: English1,{EN_FONT},{EN_SIZE},{en1_color},{en1_color},{outline},{box},-1,0,0,0,100,100,0,0,1,1,0,5,{MARGIN_L},{MARGIN_R},0,1
Style: English2,{EN_FONT},{EN_SIZE},{en2_color},{en2_color},{outline},{box},-1,0,0,0,100,100,0,0,1,1,0,5,{MARGIN_L},{MARGIN_R},0,1
Style: BoxChinese,{ZH_FONT},{ZH_SIZE},{box},{box},{box},{box},0,0,0,0,100,100,0,0,1,0,0,7,0,0,0,1
Style: BoxEnglish1,{EN_FONT},{EN_SIZE},{box},{box},{box},{box},0,0,0,0,100,100,0,0,1,0,0,7,0,0,0,1
Style: BoxEnglish2,{EN_FONT},{EN_SIZE},{box},{box},{box},{box},0,0,0,0,100,100,0,0,1,0,0,7,0,0,0,1

[Events]
Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text
"""

    lines = [header]

    for item in items:
        start = item["start"]
        end = item["end"]

        # -------------------------
        # 中文
        # -------------------------
        zh = item["zh"]

        if zh:
            width = box_width(
                zh,
                ZH_SIZE,
                ZH_CHAR_WIDTH
            )

            lines.append(
                dialogue_box(
                    0,
                    start,
                    end,
                    "BoxChinese",
                    width,
                    BOX_HEIGHT_ZH,
                    VIDEO_WIDTH // 2,
                    ZH_Y,
                )
            )

            lines.append(
                dialogue_text(
                    1,
                    start,
                    end,
                    "Chinese",
                    zh,
                    VIDEO_WIDTH // 2,
                    ZH_Y,
                )
            )

        # -------------------------
        # 英文第一部分
        # -------------------------
        en1 = item["en1"]

        if en1:
            width = box_width(
                en1,
                EN_SIZE,
                EN_CHAR_WIDTH
            )

            lines.append(
                dialogue_box(
                    0,
                    start,
                    end,
                    "BoxEnglish1",
                    width,
                    BOX_HEIGHT_EN,
                    VIDEO_WIDTH // 2,
                    EN1_Y,
                )
            )

            lines.append(
                dialogue_text(
                    1,
                    start,
                    end,
                    "English1",
                    en1,
                    VIDEO_WIDTH // 2,
                    EN1_Y,
                )
            )

        # -------------------------
        # 英文第二部分
        # -------------------------
        en2 = item["en2"]

        if en2:
            width = box_width(
                en2,
                EN_SIZE,
                EN_CHAR_WIDTH
            )

            lines.append(
                dialogue_box(
                    0,
                    start,
                    end,
                    "BoxEnglish2",
                    width,
                    BOX_HEIGHT_EN,
                    VIDEO_WIDTH // 2,
                    EN2_Y,
                )
            )

            lines.append(
                dialogue_text(
                    1,
                    start,
                    end,
                    "English2",
                    en2,
                    VIDEO_WIDTH // 2,
                    EN2_Y,
                )
            )

    output.write_text(
        "\n".join(lines),
        encoding="utf-8-sig"
    )


def main():
    if len(sys.argv) < 2:
        print(
            "用法：python make_ass_final_v3_2.py "

    cs = round(int(ms) / 10)
    if cs >= 100:
        cs, s = 0, str(int(s) + 1)

def parse_srt(text):

def split_chinese(text):
    text = text.strip()

        return "", ""
    if "\n" in text:

def split_english(text):

    if not text:

    if "\n" in text:

# ============================================================
# 五轨 SRT 拆分版
#
# Python 只负责：
#   1. 读取原 SRT
#   2. 保留原时间轴
#   3. 中文拆成 Cn1 / Cn2
#   4. 英文拆成 En1 / En2 / En3
#
# SRT 本身不写位置、字体、坐标、字幕框等信息。
# 最终位置完全交给字幕软件。

def clean_text(text):
    # 删除 HTML 标签，保留普通字幕文字
    text = re.sub(r"<[^>]+>", "", text)
    return text.strip()


def parse_srt(text):
    result = []

    blocks = re.split(r"\r?\n\s*\r?\n", text.strip())

    for block in blocks:
        lines = [
            x.strip("\ufeff").strip()
            for x in block.splitlines()
            if x.strip()
        ]

        if len(lines) < 3:
            continue

        # 找时间轴，不强制要求它一定在第二行
        time_index = None
        start = None
        end = None

        for i, line in enumerate(lines[:3]):
            m = re.match(
                r"(\d{2}:\d{2}:\d{2}[,.]\d{3})\s*-->\s*"
                r"(\d{2}:\d{2}:\d{2}[,.]\d{3})",
                line
            )

            if m:
                time_index = i
                start = m.group(1).replace(".", ",")
                end = m.group(2).replace(".", ",")
                break

        if time_index is None:
            continue

        subtitle_lines = [
            clean_text(x)
            for x in lines[time_index + 1:]
            if clean_text(x)
        ]

        if not subtitle_lines:
            continue

        # 当前双语 SRT：
        #
        # English
        # 中文
        #
        # 因此：
        # subtitle_lines[0] = English
        # subtitle_lines[1] = Chinese

        english = subtitle_lines[0]
        chinese = subtitle_lines[1] if len(subtitle_lines) > 1 else ""

        result.append({
            "start": start,
            "end": end,
            "english": english,
            "chinese": chinese,
        })

    return result


def split_chinese(text):
    """
    中文最多两行。
    尽量在标点处断开。
    """

    text = text.strip()

    if not text:
        return "", ""

    if "\n" in text:
        parts = [x.strip() for x in text.splitlines() if x.strip()]
        return (
            parts[0],
            parts[1] if len(parts) > 1 else ""
        )

    # 短句不拆
    if len(text) <= 16:
        return text, ""

    punctuation = "，。！？；：、,.!?;:"
    target = len(text) / 2

    candidates = []

    for i, char in enumerate(text):
        if char in punctuation and 0 < i < len(text) - 1:
            candidates.append(
                (abs((i + 1) - target), i + 1)
            )

    if candidates:
        position = min(candidates)[1]
    else:
        position = len(text) // 2

    return (
        text[:position].strip(),
        text[position:].strip()
    )


def split_english(text):
    """
    英文最多三行。
    按单词拆分，不在单词中间切断。
    """

    text = re.sub(r"\s+", " ", text.strip())

    if not text:
        return "", "", ""

    # 原本已有换行
    if "\n" in text:
        parts = [
            x.strip()
            for x in text.splitlines()
            if x.strip()
        ]

        parts = parts[:3]

        return (
            parts[0] if len(parts) > 0 else "",
            parts[1] if len(parts) > 1 else "",
            parts[2] if len(parts) > 2 else "",
        )

    # 比较短，不拆
    if len(text) <= 42:
        return text, "", ""

    words = text.split()

    lines = []
    current = ""

    for word in words:

        candidate = (
            word
            if not current
            else current + " " + word
        )

        if len(candidate) <= 42:
            current = candidate

        else:
            if current:
                lines.append(current)

            current = word

    if current:
        lines.append(current)

    # 最多三行
    if len(lines) > 3:
        lines = [
            lines[0],
            lines[1],
            " ".join(lines[2:])
        ]

    while len(lines) < 3:
        lines.append("")

    return lines[0], lines[1], lines[2]


def write_srt(path, rows):
    """
    写普通 SRT。
    没有任何 ASS 样式或位置数据。
    """

    output = []

    number = 1

    for start, end, text in rows:

        if not text:
            continue

        output.append(str(number))
        output.append(
            f"{start} --> {end}"
        )
        output.append(text)
        output.append("")

        number += 1

    path.write_text(
        "\n".join(output),
        encoding="utf-8-sig"
    )
